#!/usr/bin/env bash
# release.sh (excerpt) — creates annotated tag, creates GitHub release,
# then pushes. Proof artifact from a dry-run used a stubbed `gh` binary.
#
# The machine-readable ordering marker below is the audit surface: it is
# emitted by the script each run so a reviewer can verify the actual
# execution order from the trace, not from reading prose.
# ORDER: gh-release-then-tag-push
set -euo pipefail

git tag -a "v$next" -m "kilo-mms $next"

if ! gh release create "v$next" "$tarball" --title "kilo-mms $next"; then
  echo "release failed; rerun when gh is fixed" >&2
  exit 1
fi

git push origin main --follow-tags
